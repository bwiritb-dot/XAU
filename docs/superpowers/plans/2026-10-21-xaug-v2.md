**Goal:** Отдельное Chrome-расширение `XAUG-v2/`: те же расчёты, что в `app.js` (WSB — метка A, Разворот — метка B), но экран без единого слова — каждый фактор сигнала виден, отсортирован по весу и показывает, сколько баллов он дал.

**Architecture:** `tools/build-core.mjs` вырезает функции расчёта из оригинального `app.js` побайтно и собирает ES-модуль `src/core.js` — математику руками не переписываем. UI (`src/ui/*`) — чистые функции «данные → HTML-строка», поэтому тоже тестируются в Node.

**Tech Stack:** чистый JS (ES-модули), Chrome Manifest V3, TradingView Lightweight Charts v4.1.3 (уже лежит в `lib/`), Node 24 `node:test` без зависимостей.

## Global Constraints

- Оригинал (`app.js`, `index.html`, `styles.css`, `popup.*`, `background.js`, `manifest.json`, `server.py` в корне) **не меняется ни на байт**.
- `XAUG-v2/src/core.js` **только генерируется** `node tools/build-core.mjs`; правка руками запрещена.
- Математика: WSB — версия A (метка: закрытие за боксом, «процент» 68–94); Разворот — версия B (взвод за 12 свечей + 3 из 4 разворотов, «процент» 70–92).
- `trades.csv` не используется: `historicalTrades` всегда `[]`.
- На экране **ни одной буквы** (латиница и кириллица): только цифры, знаки `+ - . / % $ :`, цвет, фигуры, SVG-иконки. Исключение — `<title>` вкладки.
- Цвета: покупка `#10b981`, продажа `#f43f5e`, не сработало `#64748b`, цена/акцент `#f59e0b`, фон `#0b0f17`.
- Данные: Binance `XAUUSDT`, 15m, 1000 свечей REST + поток `wss://fstream.binance.com/ws/xauusdt@kline_15m` — как в оригинале (`CONFIG` из core.js).
- Зона смерти — `isDeathZone` оригинала (Пт 12:00 – Пн 12:00, UTC+3).
- Без npm-зависимостей. Тесты: `node --test "tests/*.test.mjs"` из папки `XAUG-v2` (Node 24: просто `tests/` не работает).
- **Облачная сессия:** Binance из облака может быть недоступен — все тесты работают только на `tests/fixtures/klines.json` (3000 свечей, 03.08–03.09.2026, уже в репозитории). Сеть в тестах не трогать.

```js
// Собирает src/core.js из оригинального app.js: функции копируются побайтно.
// Запуск: node tools/build-core.mjs
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";

const here = path.dirname(fileURLToPath(import.meta.url));
export const APP_JS = path.resolve(here, "../../app.js");
export const CORE_JS = path.resolve(here, "../src/core.js");

export const FUNCTIONS = [
  "calculateIndicators",
  "calculateATR",
  "isDeathZone",
  "classifyCandleWick",
  "calculateBreakoutProbability",
  "calculateMeanReversionProbability",
  "calculateWSBOrderFlow",
  "updateTradeMarkers",
];

// Текст от начала `head` до парной закрывающей скобки (и `;`, если он сразу за ней).
export function sliceBlock(src, head) {
  const start = src.indexOf(head);
  if (start === -1) throw new Error(`not found: ${head}`);
  const open = src.indexOf("{", start);
  let depth = 0;
  for (let i = open; i < src.length; i++) {
    if (src[i] === "{") depth++;
    else if (src[i] === "}") {
      depth--;
      if (depth === 0) {
        let end = i + 1;
        if (src[end] === ";") end++;
        return src.slice(start, end);
      }
    }
  }
  throw new Error(`unbalanced: ${head}`);
}

export function extractAll(src) {
  const blocks = { CONFIG: sliceBlock(src, "const CONFIG = {") };
  for (const name of FUNCTIONS) {
    // "\n" перед "function" — только объявления верхнего уровня
    blocks[name] = sliceBlock(src, `\nfunction ${name}(`).slice(1);
  }
  return blocks;
}

export function buildCore(src) {
  const b = extractAll(src);
  return [
    "// AUTO-GENERATED by tools/build-core.mjs from ../app.js — DO NOT EDIT BY HAND.",
    "// Функции ниже скопированы из app.js побайтно; проверка: tests/core-identity.test.mjs.",
    "let historicalTrades = [];",
    "let candleSeries = null;",
    'let activeStrategy = "wsb";',
    "let wsbCurrentState = null;",
    "",
    b.CONFIG,
    "",
    ...FUNCTIONS.map((n) => b[n] + "\n"),
    "// Оригинальный updateTradeMarkers рисует в candleSeries; здесь он отдаёт метки массивом.",
    "export function computeMarkers(data, strategy, wsbState) {",
    "  let out = [];",
    "  candleSeries = { setMarkers: (m) => { out = m; } };",
    "  activeStrategy = strategy;",
    "  historicalTrades = [];",
    "  wsbCurrentState = wsbState || null;",
    "  updateTradeMarkers(data);",
    "  candleSeries = null;",
    "  return out;",
    "}",
    "",
    `export { CONFIG, ${FUNCTIONS.join(", ")} };`,
    "",
  ].join("\n");
}

if (process.argv[1] && path.resolve(process.argv[1]) === fileURLToPath(import.meta.url)) {
  fs.mkdirSync(path.dirname(CORE_JS), { recursive: true });
  fs.writeFileSync(CORE_JS, buildCore(fs.readFileSync(APP_JS, "utf8")));
  console.log(`written ${CORE_JS}`);
}
```

```js
import fs from "node:fs";

import { APP_JS } from "../../tools/build-core.mjs";

let api = null;

export function loadOriginal() {
  if (api) return api;
  const el = new Proxy(function () {}, {

      if (k === "querySelector") return () => el;
      if (k === "querySelectorAll") return () => [];
      if (typeof k === "string" && /^(add|remove|append|set|insert|click|focus)/.test(k)) return () => {};
      return el;
    },
    set: () => true,
    apply: () => el,
  });
  const doc = { getElementById: () => el, querySelector: () => el, querySelectorAll: () => [], addEventListener: () => {}, createElement: () => el, body: el, readyState: "loading" };

    document: doc, window: { addEventListener: () => {} }, console: { log() {}, warn() {}, error() {} },
```

```js
import { CONFIG, calculateIndicators, calculateWSBOrderFlow, computeMarkers } from "./core.js";

// Разбор свечей — как в app.js (fetchHistoricalData / ws.onmessage).
export function mapRestKline(k) {
  const vol = parseFloat(k[5]);
  const takerBuy = parseFloat(k[9] || (vol * 0.5));
  return {
    time: Math.floor(k[0] / 1000),
    open: parseFloat(k[1]),
    high: parseFloat(k[2]),
    low: parseFloat(k[3]),
    close: parseFloat(k[4]),
    volume: vol,
    takerBuy: takerBuy,
    delta: 2.0 * takerBuy - vol,
  };
}

export function mapWsKline(k) {
  const vol = parseFloat(k.v);
  const takerBuy = parseFloat(k.V || (vol * 0.5));
  return {
    time: Math.floor(k.t / 1000),
    open: parseFloat(k.o),
    high: parseFloat(k.h),
    low: parseFloat(k.l),
    close: parseFloat(k.c),
    volume: vol,
    takerBuy: takerBuy,
    delta: 2.0 * takerBuy - vol,
    isClosed: k.x,
  };
}

export function applyKline(data, candle) {
  if (data.length === 0) return data;
  const lastIdx = data.length - 1;
  if (data[lastIdx].time === candle.time) data[lastIdx] = candle;
  else data.push(candle);
  return data;
}

// Порядок расчёта — как в оригинале: индикаторы → WSB → метки.
export function pipeline(data, strategy) {
  calculateIndicators(data);
  const wsb = calculateWSBOrderFlow(data);
  const markers = computeMarkers(data, strategy, wsb);
  return { data, wsb, markers };
}

    ws.onerror = () => onStatus(false);
    ws.onclose = () => { onStatus(false); if (!stopped) setTimeout(open, 3000); };
  };

  const timer = setInterval(() => {
    if (Date.now() - lastAt <= 30000) return;
    lastAt = Date.now();
    onStale();
    try { ws.onclose = null; ws.close(); } catch (e) {}
    setTimeout(open, 1000);
  }, 3000);
```

```js
const byMaxDesc = (a, b) => b.max - a.max;

// Повторяет calculateBreakoutProbability (app.js) по слагаемым. Тест: total === «процент» оригинала.
export function explainBreakout(box, bar, data, barIdx) {
  const wCount = box.wickCount || 3;
  const wicks = wCount >= 5 ? 12 : wCount >= 4 ? 8 : wCount >= 3 ? 5 : 0;

  const isLong = box.breakoutDir === "LONG";
  const dirMatch = (isLong && box.signalDir === "LONG") || (!isLong && box.signalDir === "SHORT") ? 8 : 0;

  const idx = barIdx !== undefined ? barIdx : data.findIndex((d) => d.time === bar.time);
  const fromIdx = Math.max(0, (box.startIdx || idx) - 4);
  const toIdx = Math.min(data.length - 1, idx >= 0 ? idx : data.length - 1);
  let formDelta = 0;
  let formVol = 0;
  for (let k = fromIdx; k <= toIdx; k++) {
    formDelta += (data[k].delta || 0);
    formVol += (data[k].volume || 0);
  }
  const dr = formVol > 0 ? (formDelta / formVol) : 0;
  let delta = 0;
  if (isLong) {
    if (dr > 0.20) delta = 12; else if (dr > 0.10) delta = 8; else if (dr > 0.0) delta = 4;
  } else {
    if (dr < -0.20) delta = 12; else if (dr < -0.10) delta = 8; else if (dr < -0.0) delta = 4;
  }

  let zScore = null;
  let z = 0;
  if (idx > 10) {
    const zStart = Math.max(0, idx - 60);
    let zSum = 0;
    let count = 0;
    for (let k = zStart; k < idx; k++) { zSum += (data[k].delta || 0); count++; }
    const zMean = count > 0 ? zSum / count : 0;
    let zVar = 0;
    for (let k = zStart; k < idx; k++) zVar += Math.pow((data[k].delta || 0) - zMean, 2);
    const zStd = Math.sqrt(count > 0 ? zVar / count : 1.0) || 1.0;
    zScore = ((bar.delta || 0) - zMean) / zStd;
    if (isLong && zScore >= 1.0) z = 6;
    else if (!isLong && zScore <= -1.0) z = 6;
  }

  let vwap = 0;
  if (bar.vwap) {
    if (isLong && bar.close > bar.vwap) vwap = 5;
    if (!isLong && bar.close < bar.vwap) vwap = 5;
  }

  const atr = bar.atr || 5.0;
  const range = bar.high - bar.low;
  const impulse = range >= atr * 0.7 ? 4 : 0;

  const factors = [
    { key: "wicks", max: 12, pts: wicks, value: wCount, steps: [5, 8, 12] },
    { key: "delta", max: 12, pts: delta, value: dr, steps: [4, 8, 12] },
    { key: "dirMatch", max: 8, pts: dirMatch, value: dirMatch ? 1 : 0 },
    { key: "z", max: 6, pts: z, value: zScore },
    { key: "vwap", max: 5, pts: vwap, value: bar.vwap ? bar.close - bar.vwap : null },
    { key: "impulse", max: 4, pts: impulse, value: range / atr },
  ].sort(byMaxDesc);

  const raw = 56 + wicks + dirMatch + delta + z + vwap + impulse;
  return { dir: box.breakoutDir, base: 56, raw, total: Math.min(94, Math.max(68, Math.round(raw))), clamp: [68, 94], factors };
}

export function explainMrProb(cur, prev) {
  const aoFlip = cur.aoColor !== prev.aoColor ? 7 : 0;
  const smi = Math.abs(cur.smiTsi) > 20 ? 8 : 0;
  const uo = (cur.uo < 30 || cur.uo > 70) ? 8 : 0;
  const lrs = Math.abs(cur.lrs) > 4.0 ? 6 : 0;
  const factors = [
    { key: "smi", max: 8, pts: smi, value: cur.smiTsi },
    { key: "uo", max: 8, pts: uo, value: cur.uo },
    { key: "aoFlip", max: 7, pts: aoFlip, value: cur.aoColor },
    { key: "lrs", max: 6, pts: lrs, value: cur.lrs },
  ].sort(byMaxDesc);
  const raw = 65 + aoFlip + smi + uo + lrs;
  return { base: 65, raw, total: Math.min(92, Math.max(70, Math.round(raw))), clamp: [70, 92], factors };
}

const ARM = [

  { key: "uo", w: 25, long: (b) => b.uo < 40, short: (b) => b.uo > 60 },

export function mrBarState(data, i) {
  const cur = data[i];
  const prev = data[i - 1];
  const arm = { long: ARM.map((c) => c.long(cur)), short: ARM.map((c) => c.short(cur)) };
  const win = { long: [], short: [] };
  for (let k = 11; k >= 0; k--) {
    const b = data[i - k];
    win.long.push(!!b && ARM.every((c) => c.long(b)));
    win.short.push(!!b && ARM.every((c) => c.short(b)));
  }
  const turns = prev
    ? {
        long: [cur.aoColor === 1 && prev.aoColor === -1, cur.smiTsi > cur.smiSig && prev.smiTsi <= prev.smiSig, cur.uo > prev.uo, cur.lrs > prev.lrs],
        short: [cur.aoColor === -1 && prev.aoColor === 1, cur.smiTsi < cur.smiSig && prev.smiTsi >= prev.smiSig, cur.uo < prev.uo, cur.lrs < prev.lrs],
      }
    : { long: [false, false, false, false], short: [false, false, false, false] };
  const longArmed = win.long.some(Boolean);
  const shortArmed = win.short.some(Boolean);
  const longTrigger = turns.long.filter(Boolean).length >= 3;
  const shortTrigger = turns.short.filter(Boolean).length >= 3;
  const dead = isDeathZone(cur.time);
  let entry = null;
  if (i >= 25) {
    if (longArmed && longTrigger) entry = dead ? null : "LONG";
    else if (shortArmed && shortTrigger) entry = dead ? null : "SHORT";
  }
  const sum = (flags) => ARM.reduce((a, c, k) => a + (flags[k] ? c.w : 0), 0);
  return {
    arm, keys: ARM.map((c) => c.key), weights: ARM.map((c) => c.w),
    readyLong: sum(arm.long), readyShort: sum(arm.short),
    win, turns, longArmed, shortArmed, longTrigger, shortTrigger, dead, entry,
  };
}

// Сигналы WSB — тот же отбор, что в начале updateTradeMarkers (ветка wsb).
export function wsbSignals(data, wsb) {
  if (!wsb) return [];
  const boxes = [...(wsb.allBoxes || [])];
  if (wsb.activeBox && wsb.activeBox.breakoutOccurred && !boxes.some((b) => b.startTime === wsb.activeBox.startTime)) {
    boxes.push(wsb.activeBox);
  }
  const seen = new Set();
  const out = [];
  for (const box of boxes) {
    if (!box.breakoutOccurred || !box.breakoutBar) continue;
    const bar = box.breakoutBar;
    if (isDeathZone(bar.time)) continue;
    const k = `${bar.time}_${box.breakoutDir}`;
    if (seen.has(k)) continue;
    seen.add(k);
    const prob = box.probability || calculateBreakoutProbability(box, bar, data);
    out.push({
      strategy: "wsb", idx: data.findIndex((d) => d.time === bar.time), time: bar.time,
      dir: box.breakoutDir, entry: bar.close, prob, box, explain: explainBreakout(box, bar, data, box.endIdx),
    });
  }
  return out.sort((a, b) => a.time - b.time);
}

// Сигналы Разворота берём из стрелок оригинала (точны по построению), раскладку — из правил.
export function mrSignals(data, markers) {

}

export function tradeLevels(sig) {
  const e = sig.entry;
  const L = sig.dir === "LONG";
  let sl;
  if (sig.strategy === "wsb") {
    const rd = (sig.box.breakoutBar.atr || 5.0) * 1.5;
    sl = L ? Math.min(e - rd, sig.box.bottom) : Math.max(e + rd, sig.box.top);
  } else {
    sl = L ? e * 0.990 : e * 1.0075;
  }
  return {
    entry: e, sl,
    tp1: L ? e * 1.005 : e * 0.995,
    tp2: L ? e * 1.0075 : e * 0.9925,
    tp3: L ? e * 1.010 : e * 0.990,
    be: L ? e * 1.0004 : e * 0.9996,
    p50: L ? e * 1.005 : e * 0.995,
  };
}

const TEXT = {
  TP1: "■ +0.50% (50% TP1)",
  TP2: "■ +0.75% (25% TP2)",
  TP3: "■ +1.00% (25% TP3)",
  BE: "■ [РОЙ: +0.00%] (Б/У Стоп 50%)",
  P50: "■ [РОЙ: +0.50%] (Стоп +0.5% 25%)",
};
const COLOR = { SL: "#f43f5e", BE: "#eab308", TP1: "#10b981", TP2: "#10b981", TP3: "#10b981", P50: "#10b981" };

// Сопровождение сделки — повтор обеих симуляций updateTradeMarkers. WSB: ≤150 свечей, TP-цепочка в одной свече;
// Разворот: без лимита, после TP1 на той же свече проверяется стоп Б/У.
export function tradeState(sig, data) {
  const lv = tradeLevels(sig);
  const L = sig.dir === "LONG";
  const hit = (b, p) => (L ? b.high >= p : b.low <= p);
  const stop = (b, p) => (L ? b.low <= p : b.high >= p);
  const events = [];
  let stage = 0;
  let sl = lv.sl;
  let open = true;
  const ev = (j, kind) => {
    const text = kind === "SL"
      ? `■ [РОЙ: ${(((L ? sl - lv.entry : lv.entry - sl) / lv.entry) * 100 - 0.04).toFixed(2)}%] (Stop Loss)`
      : TEXT[kind];
    events.push({ idx: j, time: data[j].time, kind, text, color: COLOR[kind] });
  };
  const wsb = sig.strategy === "wsb";
  const end = wsb ? Math.min(sig.idx + 150, data.length) : data.length;
  if (sig.idx < 0) return { levels: lv, sl, stage, events, open: false, expired: false };
  for (let j = sig.idx + 1; j < end && open; j++) {
    const b = data[j];
    if (wsb) {
      if (stage === 0) {
        if (stop(b, sl)) { ev(j, "SL"); open = false; }
        else if (hit(b, lv.tp1)) {
          ev(j, "TP1"); sl = lv.be; stage = 1;
          if (hit(b, lv.tp2)) { ev(j, "TP2"); sl = lv.p50; stage = 2; if (hit(b, lv.tp3)) { ev(j, "TP3"); open = false; } }
        }
      } else if (stage === 1) {
        if (stop(b, sl)) { ev(j, "BE"); open = false; }
        else if (hit(b, lv.tp2)) { ev(j, "TP2"); sl = lv.p50; stage = 2; if (hit(b, lv.tp3)) { ev(j, "TP3"); open = false; } }
      } else {
        if (stop(b, sl)) { ev(j, "P50"); open = false; }
        else if (hit(b, lv.tp3)) { ev(j, "TP3"); open = false; }
      }
    } else {
      if (stage === 0) {
        if (stop(b, sl)) { ev(j, "SL"); open = false; continue; }
        if (hit(b, lv.tp1)) { ev(j, "TP1"); sl = lv.be; stage = 1; }
      }
      if (stage === 1) {
        if (stop(b, sl)) { ev(j, "BE"); open = false; continue; }
        if (hit(b, lv.tp2)) { ev(j, "TP2"); sl = lv.p50; stage = 2; }
      }
      if (stage === 2) {
        if (stop(b, sl)) { ev(j, "P50"); open = false; continue; }
        if (hit(b, lv.tp3)) { ev(j, "TP3"); open = false; }
      }
    }
  }
  const expired = open && wsb && end < data.length;
  return { levels: lv, sl, stage, events, open: open && !expired, expired };
}

export function openTrade(signals, data) {
  for (let k = signals.length - 1; k >= 0; k--) {
    const state = tradeState(signals[k], data);
    if (state.open) return { sig: signals[k], state };
  }
  return null;
}
```

```js
import { ICON } from "./icons.js";

export const C = { long: "#10b981", short: "#f43f5e", off: "#64748b", gold: "#f59e0b" };

export const num = (v, d = 2) => (typeof v === "number" && Number.isFinite(v) ? v.toFixed(d) : "");
export const signed = (v, d = 2) => (typeof v === "number" && Number.isFinite(v) ? (v >= 0 ? "+" : "") + v.toFixed(d) : "");

// Значение фактора рядом со шкалой — только числа.
function factorValue(f) {
  switch (f.key) {
    case "wicks": return `${f.value}/5`;
    case "delta": return signed(f.value, 2);
    case "dirMatch": return f.value ? "=" : "≠";
    case "z": return signed(f.value, 1);
    case "vwap": return signed(f.value, 1);
    case "impulse": return num(f.value, 2);
    case "smi": return signed(f.value, 0);
    case "uo": return num(f.value, 0);
    case "aoFlip": return f.pts ? "⇅" : "·";
    case "lrs": return signed(f.value, 1);
    default: return "";
  }
}

// Полоса 0–100: база серым, заработанные баллы цветом, вне [min, max] — штриховка (зажим).
export function scoreBlock(score, color) {
  const pct = (v) => Math.max(0, Math.min(100, v));
  let x = 0;
  const seg = (w, fill, extra = "") => {
    const r = `<rect x="${pct(x)}" y="0" width="${Math.max(0, pct(x + w) - pct(x))}" height="10" fill="${fill}" ${extra}/>`;
    x += w;
    return r;
  };
  const segs = [seg(score.base, C.off, 'opacity="0.5"'), ...score.factors.filter((f) => f.pts > 0).map((f) => seg(f.pts, color))];
  const [lo, hi] = score.clamp;
  const bar = `<svg class="scorebar" viewBox="0 0 100 10" preserveAspectRatio="none">
    <rect x="0" y="0" width="100" height="10" fill="#1e293b"/>${segs.join("")}
    <rect x="0" y="0" width="${lo}" height="10" fill="url(#hatch)" opacity="0.6"/>
    <rect x="${hi}" y="0" width="${100 - hi}" height="10" fill="url(#hatch)" opacity="0.6"/>
    <line x1="${score.total}" y1="0" x2="${score.total}" y2="10" stroke="#fff" stroke-width="0.8"/>
  </svg>`;
  const unit = 12; // px за балл
  const rows = score.factors.map((f) => {
    const on = f.pts > 0;
    const ticks = (f.steps || []).slice(0, -1).map((s) => `<i class="tick" style="left:${(s / f.max) * 100}%"></i>`).join("");
    return `<div class="row ${on ? "on" : "off"}" style="--c:${on ? color : C.off}">
      <span class="ic">${ICON[f.key] || ""}</span>
      <span class="track" style="width:${f.max * unit}px"><b style="width:${(f.pts / f.max) * 100}%"></b>${ticks}</span>
      <span class="pts">${on ? "+" + f.pts : "0"}</span>
      <span class="val">${factorValue(f)}</span>
    </div>`;
  }).join("");
  return `<div class="score"><div class="total" style="color:${color}">${score.total}</div>${bar}<div class="base">${score.base}</div>${rows}</div>`;
}

// Горизонтальная шкала: пороги цветными засечками, полоса (band) серым, значение — игла.
export function gauge({ min, max, value, ticks = [], band = null }) {
  const W = 160;
  const X = (v) => ((v - min) / (max - min)) * W;
  const ok = (v) => typeof v === "number" && Number.isFinite(v) && max > min;
  const parts = [`<line x1="0" y1="8" x2="${W}" y2="8" stroke="#334155" stroke-width="2"/>`];
  if (band && ok(band[0]) && ok(band[1])) {
    parts.push(`<rect x="${X(Math.max(min, band[0]))}" y="4" width="${Math.max(0, X(Math.min(max, band[1])) - X(Math.max(min, band[0])))}" height="8" fill="${C.off}" opacity="0.6"/>`);
  }
  for (const t of ticks) if (ok(t.v)) parts.push(`<line x1="${X(t.v)}" y1="1" x2="${X(t.v)}" y2="15" stroke="${t.color}" stroke-width="2"/>`);
  if (ok(value)) {
    const v = Math.max(min, Math.min(max, value));
    parts.push(`<polygon class="needle" points="${X(v) - 4},0 ${X(v) + 4},0 ${X(v)},8" fill="#fff"/>`);
  }
  return `<svg class="gauge" viewBox="-4 0 ${W + 8} 16" width="${W + 8}" height="16">${parts.join("")}</svg>`;
}
```

```js
function miniCandles(bars) {
  if (bars.length === 0) return "";
  const hi = Math.max(...bars.map((b) => b.high));
  const lo = Math.min(...bars.map((b) => b.low));
  const Y = (p) => (hi > lo ? ((hi - p) / (hi - lo)) * 56 + 2 : 30);
  return bars.map((b) => {
    const wc = b.wickType === "bull_wick" ? C.long : b.wickType === "bear_wick" ? C.short : C.off;
    const top = Math.max(b.open, b.close);
    const bot = Math.min(b.open, b.close);
    return `<svg class="mc" viewBox="0 0 20 60" width="22" height="60">
      <line x1="10" y1="${Y(b.high)}" x2="10" y2="${Y(top)}" stroke="${b.wickType === "bear_wick" ? wc : C.off}" stroke-width="${b.wickType === "bear_wick" ? 3 : 1.5}"/>

  }).join("");

function boxView(box, close) {
  if (!box || !box.active) return `<div class="boxv empty">${ICON.box}</div>`;
```

```js
let audioCtx = null;

// Тот же звук, что playTripleAlertSound в app.js: 880 → 1174.66 → 1760 Гц.
export function playTriple() {
  try {
    const AC = window.AudioContext || window.webkitAudioContext;
    if (!AC) return;
    audioCtx = audioCtx || new AC();
    if (audioCtx.state === "suspended") audioCtx.resume();
    const now = audioCtx.currentTime;
    for (const b of [{ freq: 880, start: 0, dur: 0.08 }, { freq: 1174.66, start: 0.12, dur: 0.08 }, { freq: 1760, start: 0.24, dur: 0.12 }]) {
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = "sine";
      osc.frequency.setValueAtTime(b.freq, now + b.start);
      gain.gain.setValueAtTime(0.0001, now + b.start);
      gain.gain.exponentialRampToValueAtTime(0.3, now + b.start + 0.015);
      gain.gain.exponentialRampToValueAtTime(0.0001, now + b.start + b.dur);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start(now + b.start);
      osc.stop(now + b.start + b.dur + 0.02);
    }
  } catch (e) {
    console.warn("audio", e);
  }
}

export function shouldAlert(markers, bar, alerted) {
  if (!bar || isDeathZone(bar.time) || alerted.has(bar.time)) return false;
  const hit = markers.some((m) => m.time === bar.time && (m.shape === "arrowUp" || m.shape === "arrowDown"));
  if (hit) alerted.add(bar.time);
  return hit;
}
```

```js
export function drawBoxes(c, svg, wsb, data, mode) {
  svg.innerHTML = "";
  if (mode !== "wsb" || !wsb || data.length === 0) return;
  const boxes = (wsb.allBoxes || []).slice(-6);

  const width = svg.clientWidth;

    let x2 = ts.timeToCoordinate(box.endTime || lastTime);
    const y1 = c.candles.priceToCoordinate(box.top);
    const y2 = c.candles.priceToCoordinate(box.bottom);
```

```js
function recompute(withMarkers) {
  const r = pipeline(S.data, S.mode);
  S.wsb = r.wsb;
  if (withMarkers) {
    S.markers = r.markers;
    S.signals = S.mode === "wsb" ? wsbSignals(S.data, S.wsb) : mrSignals(S.data, S.markers);
    setMarkers(C, S.markers);
  }
}

let raf = 0;
function render() {
  if (raf) return;
  raf = requestAnimationFrame(() => {
    raf = 0;
    const last = S.data[S.data.length - 1];
    if (!last) return;
    $("price").textContent = last.close.toFixed(2);
    document.body.classList.toggle("dead", isDeathZone(last.time));
    const signal = S.signals.find((s) => s.time === S.selected) || S.signals[S.signals.length - 1] || null;
    $("panel").innerHTML = S.mode === "wsb" ? renderWsb({ data: S.data, wsb: S.wsb, signal }) : renderMr({ data: S.data, signal });
    $("ruler").innerHTML = renderRuler({ trade: openTrade(S.signals, S.data), price: last.close, width: $("ruler").clientWidth - 120 });
    drawBoxes(C, $("boxes"), S.wsb, S.data, S.mode);
    document.querySelectorAll("#mode button").forEach((b) => b.classList.toggle("on", b.dataset.mode === S.mode));
    $("sound").classList.toggle("on", S.sound);
  });
}

function setMode(mode) {
  S.mode = mode;
  S.selected = null;
  store.set("xaug2.mode", mode);
  recompute(true);
  setSeries(C, S.data, S.mode);
  render();
}

function onCandle(candle) {
  applyKline(S.data, candle);
  recompute(candle.isClosed);
  setSeries(C, S.data, S.mode);
  if (candle.isClosed && S.sound && shouldAlert(S.markers, S.data[S.data.length - 1], S.alerted)) playTriple();
  render();
}

async function boot() {
  $("mode").querySelector('[data-mode="wsb"]').innerHTML = ICON.box;
  $("mode").querySelector('[data-mode="mr"]').innerHTML = ICON.wave;
  $("sound").innerHTML = ICON.sound;
  $("mode").addEventListener("click", (e) => { const b = e.target.closest("button"); if (b) setMode(b.dataset.mode); });
  $("sound").addEventListener("click", () => { S.sound = !S.sound; store.set("xaug2.sound", S.sound ? "1" : "0"); render(); });
  C.chart.subscribeClick((p) => { if (p.time && S.signals.some((s) => s.time === p.time)) { S.selected = p.time; render(); } });
  C.chart.timeScale().subscribeVisibleTimeRangeChange(() => drawBoxes(C, $("boxes"), S.wsb, S.data, S.mode));

  const q = new URLSearchParams(location.search);
  if (q.get("fixture")) {
    const raw = await (await fetch("tests/fixtures/klines.json")).json();
    const end = Number(q.get("end")) || raw.length;
    S.data = raw.slice(Math.max(0, end - 1000), end).map(mapRestKline);
  } else {
    S.data = await fetchKlines();
  }
  recompute(true);
  setSeries(C, S.data, S.mode);
  render();
  if (q.get("fixture")) return;
  connectStream({
    onCandle,
    onStatus: (on) => $("live").classList.toggle("on", on),
    onStale: async () => {
      const fresh = await fetchKlines();
      if (fresh.length) { S.data = fresh; recompute(true); setSeries(C, S.data, S.mode); render(); }
    },
  });
}

boot();
```

Примечание для исполнителя: `recompute(false)` всё равно пересчитывает индикаторы и WSB (как оригинал на каждом тике), но метки — только на закрытой свече (как `isMarkersVisible`-ветка оригинала). REST-обновление по сторожу пересчитывает и метки — математику это не меняет.
